cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" NETCOLLECT_VERSION LIMIT_COUNT 1)
if(NOT NETCOLLECT_VERSION MATCHES "^V([0-9]+)\\.([0-9]+)$")
  message(FATAL_ERROR "VERSION must use the form V<major>.<minor>")
endif()
set(NETCOLLECT_SEMVER "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.0")
project(NetCollcet-C VERSION ${NETCOLLECT_SEMVER} LANGUAGES C)
configure_file(csrc/version.h.in generated/version.h @ONLY)
if(CMAKE_C_COMPILER MATCHES "zig.exe$")
  set(CMAKE_C_ARCHIVE_CREATE "\"${CMAKE_C_COMPILER}\" ar qc <TARGET> <LINK_FLAGS> <OBJECTS>")
  set(CMAKE_C_ARCHIVE_FINISH "\"${CMAKE_C_COMPILER}\" ranlib <TARGET>")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "This application uses the Windows native desktop and encoding APIs.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CRYPTO_BACKEND WinCNG CACHE STRING "" FORCE)
set(ENABLE_ZLIB_COMPRESSION OFF CACHE BOOL "" FORCE)
set(ENABLE_ECDSA_WINCNG ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/libssh2 EXCLUDE_FROM_ALL)
set(EXPAT_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(EXPAT_DTD OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/libexpat/expat EXCLUDE_FROM_ALL)
add_library(collector_core STATIC csrc/common.c csrc/log.c csrc/inventory.c csrc/ssh.c csrc/telnet_transport.c
  third_party/libtelnet/libtelnet.c
  third_party/miniz/miniz.c third_party/miniz/miniz_zip.c
  third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_tdef.c)
target_include_directories(collector_core PUBLIC csrc "${CMAKE_CURRENT_BINARY_DIR}/generated" third_party/miniz third_party/libtelnet)
target_compile_definitions(collector_core PUBLIC UNICODE _UNICODE
  LIBSSH2_STATIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
target_link_libraries(collector_core PUBLIC libssh2_static expat ws2_32 bcrypt crypt32)
if(MSVC)
  target_compile_options(collector_core PRIVATE /utf-8 /W4)
else()
  target_compile_options(collector_core PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
add_executable(netcollect WIN32 csrc/main.c csrc/gui.c)
target_compile_definitions(netcollect PRIVATE NETCOLLECT_GUI=1)
if(NOT MSVC)
  target_link_options(netcollect PRIVATE "-Wl,--subsystem,windows")
endif()
target_link_libraries(netcollect PRIVATE collector_core comctl32 comdlg32 shell32 ole32 uuid)
add_executable(netcollect-cli csrc/main.c csrc/gui.c)
target_link_libraries(netcollect-cli PRIVATE collector_core comctl32 comdlg32 shell32 ole32 uuid)
if(MSVC)
  target_compile_options(netcollect PRIVATE /utf-8 /W4)
  target_compile_options(netcollect-cli PRIVATE /utf-8 /W4)
else()
  target_compile_options(netcollect PRIVATE -Wall -Wextra -Wno-unused-parameter)
  target_compile_options(netcollect-cli PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
add_executable(core_tests ctests/core_tests.c)
target_link_libraries(core_tests PRIVATE collector_core)
add_executable(log_tests ctests/log_tests.c)
target_link_libraries(log_tests PRIVATE collector_core)
add_executable(inspect_inventory ctests/inspect_inventory.c)
target_link_libraries(inspect_inventory PRIVATE collector_core shell32)
add_executable(gui_tests ctests/gui_tests.c)
target_link_libraries(gui_tests PRIVATE collector_core comctl32 comdlg32 shell32 ole32 uuid gdi32)
if(MSVC)
  target_compile_options(core_tests PRIVATE /utf-8)
  target_compile_options(log_tests PRIVATE /utf-8)
  target_compile_options(gui_tests PRIVATE /utf-8)
endif()
